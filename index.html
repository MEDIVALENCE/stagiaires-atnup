function setupSig(id) { const canvas = document.getElementById(id); const ctx = canvas.getContext('2d'); // Ajuste la taille réelle du dessin à la taille affichée canvas.width = canvas.offsetWidth; canvas.height = canvas.offsetHeight; let drawing = false; const getP = e => { const rect = canvas.getBoundingClientRect(); const touch = e.touches ? e.touches[0] : e; return { x: touch.clientX - rect.left, y: touch.clientY - rect.top }; }; const start = e => { drawing = true; ctx.beginPath(); const p = getP(e); ctx.moveTo(p.x, p.y); if (e.cancelable) e.preventDefault(); }; const move = e => { if (!drawing) return; const p = getP(e); ctx.lineTo(p.x, p.y); ctx.lineWidth = 2; ctx.lineCap = "round"; ctx.stroke(); if (e.cancelable) e.preventDefault(); }; const stop = () => { drawing = false; }; // Événements Tactiles (Mobile) canvas.addEventListener('touchstart', start, { passive: false }); canvas.addEventListener('touchmove', move, { passive: false }); canvas.addEventListener('touchend', stop); // Événements Souris (PC) canvas.addEventListener('mousedown', start); canvas.addEventListener('mousemove', move); window.addEventListener('mouseup', stop); }
